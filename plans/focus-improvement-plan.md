**Пример кода (TypeScript/React):**
```tsx
// src/components/transitions/ContextualTransition.tsx
import { motion, AnimatePresence } from 'framer-motion';

interface ContextualTransitionProps {
  isActive: boolean;
  direction: 'chat-to-call' | 'call-to-board' | 'board-to-chat';
  children: React.ReactNode;
}

export function ContextualTransition({ isActive, direction, children }: ContextualTransitionProps) {
  const variants = {
    'chat-to-call': {
      initial: { scale: 0.8, rotateY: -90 },
      animate: { scale: 1, rotateY: 0 },
      exit: { scale: 0.8, rotateY: 90 }
    },
    // ... другие направления
  };

  return (
    <AnimatePresence mode="wait">
      {isActive && (
        <motion.div
          variants={variants[direction]}
          initial="initial"
          animate="animate"
          exit="exit"
          transition={{ duration: 0.6, ease: "easeInOut" }}
          className="fixed inset-0 z-50"
        >
          {children}
        </motion.div>
      )}
    </AnimatePresence>
  );
}
```

**Пример кода:**
```tsx
// src/hooks/useContextualAI.ts
import { useCallback, useEffect } from 'react';
import { useAIAssistant } from './useAIAssistant';

export function useContextualAI() {
  const { suggestAction, analyzeContext } = useAIAssistant();

  const getContextualSuggestions = useCallback(async (currentPage: string, userActions: string[]) => {
    const context = await analyzeContext(currentPage, userActions);
    return context.suggestions.map(suggestion => ({
      ...suggestion,
      confidence: suggestion.confidence,
      trigger: suggestion.trigger
    }));
  }, [analyzeContext]);

  return { getContextualSuggestions, suggestAction };
}
```

**Пример кода:**
```tsx
// src/components/web3/AchievementNFT.tsx
import { useContractWrite, usePrepareContractWrite } from 'wagmi';
import { parseEther } from 'viem';

interface AchievementNFTProps {
  achievement: {
    id: string;
    title: string;
    description: string;
    rarity: 'common' | 'rare' | 'epic' | 'legendary';
    metadata: {
      image: string;
      attributes: Record<string, any>;
    };
  };
}

export function AchievementNFT({ achievement }: AchievementNFTProps) {
  const { config } = usePrepareContractWrite({
    address: '0x...', // Achievement contract address
    abi: [...], // Contract ABI
    functionName: 'mintAchievement',
    args: [achievement.id, achievement.metadata],
    value: parseEther('0.01'), // Minting fee
  });

  const { write, isLoading } = useContractWrite(config);

  return (
    <div className="achievement-nft-card">
      <div className="rarity-border" data-rarity={achievement.rarity}>
        <img src={achievement.metadata.image} alt={achievement.title} />
        <h3>{achievement.title}</h3>
        <p>{achievement.description}</p>
        <button
          onClick={() => write?.()}
          disabled={isLoading}
          className="mint-button"
        >
          {isLoading ? 'Minting...' : 'Mint NFT'}
        </button>
      </div>
    </div>
  );
}
```
